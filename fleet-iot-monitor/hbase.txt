import json
import happybase
from kafka import KafkaConsumer

# 1. الاتصال بقاعدة بيانات HBase (عبر بروتوكول Thrift على بورت 9090 أو البورت الخاص بها في الـ Docker)
# ملاحظة: سنحتاج للتأكد من تشغيل Thrift Server، أو سنربطه بالاي بي المناسب
connection = happybase.Connection('localhost', port=9090)
table_name = 'fleet_telemetry'

# إنشاء الجدول في HBase لو مش موجود (له Family باسم 'd' للبيانات)
try:
    connection.create_table(
        table_name,
        {'d': dict()}  # Column Family
    )
    print(f"📦 Table '{table_name}' created successfully in HBase.")
except Exception as e:
    print(f"ℹ️ Table might already exist or connected: {e}")

table = connection.table(table_name)

# 2. الاتصال بـ Kafka لقراءة البيانات
consumer = KafkaConsumer(
    'fleet-telemetry',
    bootstrap_servers=['localhost:9092'],
    auto_offset_reset='earliest',
    enable_auto_commit=True,
    group_id='hbase-consumer-group',
    value_deserializer=lambda x: json.loads(x.decode('utf-8')) if x else None
)

print("📥 Listening to Kafka and saving to HBase...")

try:
    for message in consumer:
        data = message.value
        device_id = data.get("device_id")
        timestamp = data.get("timestamp")
        
        # تصميم مفتاح الجدول (Row Key) يكون مميز بكل شاحنة ووقت
        row_key = f"{device_id}#{timestamp}"
        
        # تجهيز البيانات للتخزين في HBase (بصيغة bytes)
        row_data = {
            b'd:fleet_id': data.get("identity", {}).get("fleet_id", "").encode('utf-8'),
            b'd:driver_name': data.get("identity", {}).get("driver_name", "").encode('utf-8'),
            b'd:cargo_temp_c': str(data.get("telemetry", {}).get("cargo_temp_c", "")).encode('utf-8'),
            b'd:humidity_pct': str(data.get("telemetry", {}).get("humidity_pct", "")).encode('utf-8'),
            b'd:speed_kmh': str(data.get("operations", {}).get("speed_kmh", "")).encode('utf-8'),
            b'd:door_status': data.get("operations", {}).get("door_status", "").encode('utf-8')
        }
        
        # حفظ البيانات في جدول HBase
        table.put(row_key.encode('utf-8'), row_data)
        print(f"💾 Saved to HBase -> RowKey: {row_key}")

except KeyboardInterrupt:
    print("\n🛑 HBase Consumer stopped.")
    connection.close()
